// index.js - ฉบับสมบูรณ์ (Deep Link Support) require('dotenv').config(); // โหลด Environment Variables const express = require('express'); const webpush = require('web-push'); const bodyParser = require('body-parser'); const cors = require('cors'); const app = express(); // อนุญาตให้ GitHub Pages เรียกใช้งานได้ app.use(cors()); app.use(bodyParser.json()); // --- 1. การตั้งค่า VAPID KEYS --- const publicVapidKey = process.env.PUBLIC_VAPID_KEY; const privateVapidKey = process.env.PRIVATE_VAPID_KEY; const mailtoEmail = process.env.MAILTO_EMAIL || 'mailto:oonllos@gmail.com'; // ตรวจสอบค่า Key if (!publicVapidKey || !privateVapidKey) { console.error("❌ ERROR: ไม่พบ VAPID Keys กรุณาเช็ค Environment Variables ใน Render"); } else { console.log("✅ VAPID Keys Loaded."); } webpush.setVapidDetails( mailtoEmail, publicVapidKey, privateVapidKey ); // ตัวแปรเก็บ Subscription (ใน RAM) let subscriptions = []; // --- 2. Routes --- // เช็คสถานะ Server app.get('/', (req, res) => { res.send('Makro Push Server is Running (Deep Link Ready)!'); }); // รับการลงทะเบียน (Subscribe) app.post('/subscribe', (req, res) => { const subscription = req.body; subscriptions.push(subscription); console.log(`➕ ผู้ใช้ใหม่ลงทะเบียน (Total: ${subscriptions.length})`); res.status(201).json({}); }); // ยิงแจ้งเตือน (Trigger Push) app.post('/trigger-push', (req, res) => { const { message, branch } = req.body; // สร้างข้อมูลแจ้งเตือนพร้อมลิ้งค์ (Deep Link) const notificationPayload = JSON.stringify({ title: `⚡ งานด่วน! สาขา ${branch || 'ไม่ระบุ'}`, body: message || 'มีตำแหน่งงานว่าง รีบสมัครด่วน!', icon: 'https://cdn-icons-png.flaticon.com/512/3135/3135715.png', data: { url: '/?mode=apply' // รหัสลับสำหรับเปิดหน้าสมัครงาน } }); console.log(`🚀 กำลังส่งแจ้งเตือนไปยัง ${subscriptions.length} คน...`); Promise.all(subscriptions.map(sub => webpush.sendNotification(sub, notificationPayload) .catch(err => { // ถ้าส่งไม่ผ่าน (เช่น User ปิด Browser ไปแล้ว) ก็แค่ข้ามไป console.error("ส่งไม่ผ่าน 1 คน:", err.statusCode); return null; }) )) .then(() => { console.log("✅ ส่งสำเร็จทั้งหมด"); res.json({ success: true, count: subscriptions.length }); }) .catch(err => { console.error("❌ เกิดข้อผิดพลาด:", err); res.status(500).json({ error: 'Failed to send notifications' }); }); }); // --- 3. Start Server --- const port = process.env.PORT || 5000; app.listen(port, () => { console.log(`Server started on port ${port}`); });
